```python
# phase11_gui_diag.py -- GUI-runnable diagnostic for Phase 11 alt-conf
# cartoon/ribbon hider. Run in a REAL Windows PyMOL GUI session:
#   1. Launch PyMOL (setenv.bat -> pymol)
#   2. In the PyMOL command line: run smoke/phase11_gui_diag.py
#   3. Follow the printed checklist (observe the viewer + click).
# Pure pymol.cmd.* (NO Qt). The HUMAN is the oracle for the GUI-only
# failure modes headless smoke cannot see (05-08 lesson).
import sys
from pymol import cmd
from biochemeleon import game, mutation, generators

print("=" * 60)
print("Phase 11 GUI diagnostic — alt-conf cartoon/ribbon hider")
print("=" * 60)

# --- SETUP: 1ubq, collapse, build 1 cartoon alt-conf hider ---
cmd.fetch("1ubq", async_=0)
obj = "1ubq"
mutation.collapse_to_single_state(obj)
orig_count = cmd.count_atoms(obj)
print("Loaded 1ubq, collapsed to single state, orig_count=%d" % orig_count)

# Build 1 cartoon alt-conf hider (mid-chain segment, 1.5A displacement)
cas = []
cmd.iterate("%s and polymer and name CA" % obj, "stored.append((chain, resv, ID))",
            space={'stored': cas})
cas_by = {}
for ch, ri, cid in cas:
    cas_by.setdefault(ch, []).append((ri, cid))
segments = generators.pick_segments(cas_by, 1)
disp = generators.generate_middle_displacement(1, seed=42, magnitude=1.5)[0]
chain, start_resi, end_resi = segments[0]
hider_specs = [((chain, start_resi, end_resi, disp), 'cartoon')]
gc = game.GameController(obj)
gc.start(hider_specs)
cmd.show("cartoon", "%s and segi GAME" % obj)
print("Inserted 1 cartoon alt-conf hider: chain=%s resi %d-%d, middle displaced ~1.5A"
      % (chain, start_resi, end_resi))
print("Anchor (clickable) middle CA: resi %d" % (start_resi + 1))

# --- SETUP: a 2nd hider for multi-hider + multi-state display ---
segments2 = generators.pick_segments(cas_by, 2)
if len(segments2) == 2:
    disps2 = generators.generate_middle_displacement(2, seed=42)
    specs2 = [((s[0], s[1], s[2], d), 'cartoon') for s, d in zip(segments2, disps2)]
    # (re-start with 2 hiders — replace the single-hider gc)
    gc.cleanup()
    gc2 = game.GameController(obj)
    gc2.start(specs2)
    cmd.show("cartoon", "%s and segi GAME" % obj)
    print("Re-started with 2 cartoon alt-conf hiders (multi-hider + all_states).")
else:
    print("WARN: could not pick 2 disjoint segments on this chain; multi-hider check skipped.")
    gc2 = gc

# --- OBSERVATION CHECKLIST (the human verifies each) ---
print("\n" + "=" * 60)
print("OBSERVATION CHECKLIST — verify each in the viewer:")
print("=" * 60)
print("1. [SUCCESS CRIT 1] The cartoon hider renders as a CONNECTED part")
print("   of the cartoon trace (a small bulge/loop mid-chain), NOT a")
print("   disconnected segment floating away. Look for a slight bump on")
print("   the tube at resi %d-%d." % (start_resi, end_resi))
print("   EXPECT: connected tube with a displaced middle bump.")
print("   (05-08 failure: disconnected look — this fixes it.)")
print("")
print("2. [Pitfall 5 / Bug 3] The view did NOT auto-zoom into the inserted")
print("   segment on Start. The camera stayed at the full-molecule view.")
print("   EXPECT: no sudden zoom-in. (Headless is blind to this.)")
print("")
print("3. [Open Risk 6] The middle bump is DISPLACED (~1.5A off the real")
print("   trace) but the endpoints connect smoothly to the real trace.")
print("   EXPECT: a visible bulge, ends blended, middle nudged sideways.")
print("")
print("4. [SUCCESS CRIT 3] If 2 hiders were inserted, BOTH bumps are")
print("   visible (all_states is on). Rotate the view — both mid-chain")
print("   bulges should be present, not just one.")
print("   EXPECT: 2 separate bulges on the chain.")
print("")
print("5. [Open Risk 11] Click a MIDDLE atom of the bump (resi %d, any atom"
print("   — CA/N/C/O). It should register FOUND (recolor the bump, not")
print("   the real trace). Click an ENDPOINT (resi %d or %d) — should be")
print("   a Miss. Click the REAL trace (alt='') at the same residue —")
print("   should be a Miss."
print("   EXPECT: middle click = found+recolor bump; endpoint/real = miss.")
print("   (Enable PickWizard via the plugin Game tab, or use the test"
print("   below.)" % (start_resi + 1, start_resi, end_resi))
print("")
print("6. [Pitfall 14] When found, ONLY the bump (segi GAME middle) turns")
print("   the found color — the real trace stays its original color.")
print("   EXPECT: bump recolored, real trace unchanged.")
print("")
print("7. [SUCCESS CRIT 5] Press Cleanup (plugin Setup tab) — the bump")
print("   vanishes, the molecule returns to its original state (count")
print("   back to %d). EXPECT: clean restore, no residual bulge." % orig_count)
print("")
print("8. [SUCCESS CRIT 6] After Cleanup, press Start again (New Game) —")
print("   a fresh hider appears with NO corruption from the prior round.")
print("   EXPECT: clean new round.")
print("")
print("9. [SUCCESS CRIT 2 + 4] Repeat with ribbon + mixed reps via the")
print("   plugin Setup tab (set per_rep cartoon+ribbon+spheres+sticks).")
print("   EXPECT: hiders in each rep, all visible + clickable.")
print("")
print("When done, type 'approved' (or describe issues) in the chat.")
print("=" * 60)

# Keep the object loaded for the user to interact with.
# Do NOT cleanup here — the user observes + clicks + cleans up via the GUI.
```
